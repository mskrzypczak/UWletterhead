% no page number in a document of one page. Checks also where the header,
% the footer and the body text are put on the page.

	{Wydzia{\l} Matematyki, Informatyki i Mechaniki}

	{ul. Banacha~2, 02-097 Warszawa, Polska}
\CheckEqual{the prefix of the e-mail}{\UWprefEMail}{e-mail: }
\CheckEqual{the space below the header}{\UWheadsep}{10mm}
\CheckEqual{the line spacing}{\baselinestretch}{1.3}

	{\ifcsname ver@polski.sty\endcsname loaded\else missing\fi}{loaded}

% its baseline 53 mm from the top edge.

% lowest point of the header (here the depth of its only line) by
% \UWheadsep - 6.5mm and by the usual distance of two paragraphs.
