% The options english, dejavu and 11pt in a document of several pages.
% Checks the lines of the header and of the footer (an empty value removes
% its line), the page numbers, the space below the header, and the commands
% \UWdate, \UWaddressee and \UWsignature.
\documentclass[english,dejavu,11pt]{UWletterhead}
\input{tests/checks}

\CheckEqual{the logo of the University}{\UWlogo}{UWlogoEN.pdf}
\CheckEqual{the faculty}{\UWfaculty}
	{Faculty of Mathematics, Informatics and Mechanics}
\CheckEqual{the address}{\UWaddress}{Banacha~2, 02-097 Warsaw, Poland}
\CheckEqual{the package polski}
	{\ifcsname ver@polski.sty\endcsname loaded\else missing\fi}{missing}

% Every line of the header checks the colour and the weight of its text:
% blue for the units, gray for the person, bold for the first of each. The
% size is 12pt in the header and 8pt in the footer, whatever the option.
\renewcommand{\UWfaculty}{\PageMark{faculty}\CheckColour{.}{0,146,206}%
	\CheckEqual{the weight of the faculty}{\FontSeries}{b}%
	\CheckEqual{the size of the header}{\FontSize}{12}Faculty}
\renewcommand{\UWinstitute}{\PageMark{institute}\CheckColour{.}{0,146,206}%
	\CheckEqual{the weight of the institute}{\FontSeries}{m}Institute}
\renewcommand{\UWfunction}{\PageMark{function}\CheckColour{.}{128,130,133}%
	\CheckEqual{the weight of the function}{\FontSeries}{b}Function}
\renewcommand{\UWperson}{\PageMark{person}\CheckColour{.}{128,130,133}%
	\CheckEqual{the weight of the person}{\FontSeries}{m}Person}
\renewcommand{\UWprefAddress}{\PageMark{address}%
	\CheckEqual{the size of the footer}{\FontSize}{8}}
\renewcommand{\UWprefPhone}{\PageMark{phone}}
\renewcommand{\UWprefEMail}{\PageMark{email}}
\renewcommand{\UWphone}{+48 22 55 44 460}
\renewcommand{\UWemail}{iinf@mimuw.edu.pl}
\renewcommand{\thepage}{\arabic{page}\PageMark{number}}
